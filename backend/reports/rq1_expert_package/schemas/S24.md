Normalized university enrollment management system database schema in 3NF.

TABLE students
  - id (INTEGER)  [PK, UQ, NN]
  - student_code (TEXT)  [UQ, NN]
  - name (TEXT)  [NN]
  - email (TEXT)  [UQ, NN]

TABLE courses
  - id (INTEGER)  [PK, UQ, NN]
  - code (TEXT)  [UQ, NN]
  - title (TEXT)  [NN]
  - credits (INTEGER)  [NN]

TABLE enrollments
  - id (INTEGER)  [PK, UQ, NN]
  - student_id (INTEGER)  [NN, FK->students.id]
  - course_id (INTEGER)  [NN, FK->courses.id]
  - semester (TEXT)  [NN]
  - grade (TEXT)  [NN]

RELATIONSHIPS
  - enrollments.student_id [one_to_many] students.id
  - enrollments.course_id [one_to_many] courses.id
