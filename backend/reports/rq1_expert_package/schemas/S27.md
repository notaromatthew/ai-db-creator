Ospedale privato: pazienti, medici, reparti, appuntamenti, trattamenti, farmaci, prescrizioni e fatture.

TABLE patients
  - patient_id (INTEGER)  [PK, UQ, NN]
  - fiscal_code (TEXT)  [UQ, NN]
  - full_name (TEXT)  [NN]
  - birth_date (DATE)  [NN]

TABLE doctors
  - doctor_id (INTEGER)  [PK, UQ, NN]
  - full_name (TEXT)  [NN]
  - specialty (TEXT)  [NN]
  - department (TEXT)  [NN]

TABLE wards
  - ward_id (INTEGER)  [PK, UQ, NN]
  - name (TEXT)  [UQ, NN]
  - department (TEXT)  [UQ, NN]

TABLE appointments
  - appointment_id (INTEGER)  [PK, UQ, NN]
  - patient_id (INTEGER)  [NN, FK->patients.patient_id]
  - doctor_id (INTEGER)  [NN, FK->doctors.doctor_id]
  - ward_id (INTEGER)  [NN, FK->wards.ward_id]
  - scheduled_on (DATETIME)  [NN]

TABLE treatments
  - treatment_id (INTEGER)  [PK, UQ, NN]
  - appointment_id (INTEGER)  [NN, FK->appointments.appointment_id]
  - description (TEXT)  [NN]

TABLE medications
  - medication_id (INTEGER)  [PK, UQ, NN]
  - name (TEXT)  [UQ, NN]
  - active_ingredient (TEXT)  [NN]

TABLE prescriptions
  - prescription_id (INTEGER)  [PK, UQ, NN]
  - treatment_id (INTEGER)  [NN, FK->treatments.treatment_id]
  - medication_id (INTEGER)  [NN, FK->medications.medication_id]
  - dosage (TEXT)  [NN]

TABLE invoices
  - invoice_id (INTEGER)  [PK, UQ, NN]
  - appointment_id (INTEGER)  [NN, FK->appointments.appointment_id]
  - amount (REAL)  [NN]
  - issued_on (DATE)  [NN]

RELATIONSHIPS
  - appointments.patient_id [one_to_many] patients.patient_id
  - appointments.doctor_id [one_to_many] doctors.doctor_id
  - appointments.ward_id [one_to_many] wards.ward_id
  - treatments.appointment_id [one_to_many] appointments.appointment_id
  - prescriptions.treatment_id [many_to_many] treatments.treatment_id
  - prescriptions.medication_id [many_to_many] medications.medication_id
  - invoices.appointment_id [one_to_many] appointments.appointment_id
