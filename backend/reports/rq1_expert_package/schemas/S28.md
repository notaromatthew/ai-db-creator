Schema di database normalizzato in 3NF per un sistema di gestione di una biblioteca che gestisce categorie, libri, iscritti, prestiti e multe.

TABLE categories
  - category_id (INTEGER)  [PK, NN]
  - category_name (TEXT)  [UQ, NN]

TABLE books
  - book_id (INTEGER)  [PK, NN]
  - isbn (TEXT)  [UQ, NN]
  - title (TEXT)  [NN]
  - author (TEXT)  [NN]
  - category_id (INTEGER)  [NN, FK->categories.category_id]

TABLE members
  - member_id (INTEGER)  [PK, NN]
  - full_name (TEXT)  [NN]
  - email (TEXT)  [UQ, NN]
  - joined_on (DATE)  [NN]

TABLE loans
  - loan_id (INTEGER)  [PK, NN]
  - member_id (INTEGER)  [NN, FK->members.member_id]
  - book_id (INTEGER)  [NN, FK->books.book_id]
  - loan_date (DATE)  [NN]
  - due_date (DATE)  [NN]

TABLE fines
  - fine_id (INTEGER)  [PK, NN]
  - loan_id (INTEGER)  [UQ, NN, FK->loans.loan_id]
  - amount (REAL)  [NN]
  - payment_date (DATE)

RELATIONSHIPS
  - categories.category_id [one_to_many] books.category_id
  - members.member_id [one_to_many] loans.member_id
  - books.book_id [one_to_many] loans.book_id
  - loans.loan_id [one_to_one] fines.loan_id
