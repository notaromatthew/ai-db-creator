Biblioteca pubblica: categorie, libri, iscritti, prestiti e multe.

TABLE categories
  - category_id (INTEGER)  [PK, UQ, NN]
  - name (TEXT)  [UQ, NN]

TABLE members
  - member_id (INTEGER)  [PK, UQ, NN]
  - full_name (TEXT)  [NN]
  - email (TEXT)  [UQ, NN]
  - joined_on (DATE)  [NN]

TABLE books
  - book_id (INTEGER)  [PK, UQ, NN]
  - isbn (TEXT)  [UQ, NN]
  - title (TEXT)  [NN]
  - author (TEXT)  [NN]
  - category_id (INTEGER)  [NN, FK->categories.category_id]

TABLE loans
  - loan_id (INTEGER)  [PK, UQ, NN]
  - book_id (INTEGER)  [NN, FK->books.book_id]
  - member_id (INTEGER)  [NN, FK->members.member_id]
  - loan_date (DATE)  [NN]
  - due_date (DATE)  [NN]

TABLE fines
  - fine_id (INTEGER)  [PK, UQ, NN]
  - loan_id (INTEGER)  [NN, FK->loans.loan_id]
  - amount (REAL)  [NN]
  - paid_on (DATE)

RELATIONSHIPS
  - books.category_id [one_to_many] categories.category_id
  - loans.member_id [one_to_many] members.member_id
  - loans.book_id [one_to_many] books.book_id
  - fines.loan_id [one_to_many] loans.loan_id
