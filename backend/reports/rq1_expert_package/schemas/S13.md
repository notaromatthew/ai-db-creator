Normalized 3NF database schema for a hospital management system including patients, doctors, wards, medications, appointments, treatments, prescriptions, and invoices.

TABLE wards
  - ward_id (INTEGER)  [PK, NN]
  - ward_name (TEXT)  [UQ, NN]

TABLE departments
  - department_id (INTEGER)  [PK, NN]
  - department_name (TEXT)  [UQ, NN]
  - ward_id (INTEGER)  [NN, FK->wards.ward_id]

TABLE doctors
  - doctor_id (INTEGER)  [PK, NN]
  - doctor_name (TEXT)  [NN]
  - department_id (INTEGER)  [NN, FK->departments.department_id]

TABLE patients
  - patient_id (INTEGER)  [PK, NN]
  - fiscal_code (TEXT)  [UQ, NN]
  - full_name (TEXT)  [NN]
  - birth_date (DATE)  [NN]

TABLE appointments
  - appointment_id (INTEGER)  [PK, NN]
  - patient_id (INTEGER)  [NN, FK->patients.patient_id]
  - doctor_id (INTEGER)  [NN, FK->doctors.doctor_id]
  - ward_id (INTEGER)  [NN, FK->wards.ward_id]
  - scheduled_on (TEXT)  [NN]

TABLE treatments
  - treatment_id (INTEGER)  [PK, NN]
  - appointment_id (INTEGER)  [NN, FK->appointments.appointment_id]
  - treatment_name (TEXT)  [NN]
  - fee (REAL)  [NN]

TABLE medications
  - medication_id (INTEGER)  [PK, NN]
  - medication_name (TEXT)  [UQ, NN]

TABLE prescriptions
  - prescription_id (INTEGER)  [PK, NN]
  - treatment_id (INTEGER)  [NN, FK->treatments.treatment_id]
  - medication_id (INTEGER)  [NN, FK->medications.medication_id]
  - dosage (TEXT)  [NN]

TABLE invoices
  - invoice_id (INTEGER)  [PK, NN]
  - appointment_id (INTEGER)  [NN, FK->appointments.appointment_id]
  - amount (REAL)  [NN]
  - issue_date (DATE)  [NN]

RELATIONSHIPS
  - wards.ward_id [one_to_many] departments.ward_id
  - departments.department_id [one_to_many] doctors.department_id
  - patients.patient_id [one_to_many] appointments.patient_id
  - doctors.doctor_id [one_to_many] appointments.doctor_id
  - wards.ward_id [one_to_many] appointments.ward_id
  - appointments.appointment_id [one_to_many] treatments.appointment_id
  - treatments.treatment_id [one_to_many] prescriptions.treatment_id
  - medications.medication_id [one_to_many] prescriptions.medication_id
  - appointments.appointment_id [one_to_many] invoices.appointment_id
